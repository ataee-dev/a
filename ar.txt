"""
Aureon Execution Engine v2 -- the "Computer Execution Layer" (hands and feet)
=============================================================================

Receives JSON "Aureon Action Contracts", validates them strictly, performs the
requested mouse / keyboard / window / file operation, verifies the visual
outcome, and returns a structured JSON report.

Requirements
------------
    Python 3.9+
    pip install pyautogui pynput Pillow ImageHash pyperclip pygetwindow psutil

Platform notes
--------------
* Windows is the primary target. DPI awareness, multi-monitor lookup and
  active-process detection use Win32 APIs through ctypes.
* ``pygetwindow`` only supports window control on Windows (and partially
  macOS). Where it is unavailable, ``window.*`` actions and ``requires_focus``
  fail closed with a clear error instead of silently skipping the check.
* On macOS, grant Accessibility and Screen Recording permission to the host.

Action contract (input)
-----------------------
    {
      "action_id":   "act_002",                 # [A-Za-z0-9_.-], max 64 chars
      "type":        "keyboard.type",
      "params":      {...},                     # per-action schema, see below
      "constraints": {...},                     # see CONSTRAINT_SCHEMA
      "metadata":    {...}                      # free-form, never executed
    }

Unknown top-level fields, params and constraints are REJECTED. This is
deliberate: a misspelled constraint such as ``require_focus`` must never be
silently ignored.

Coordinates are normalised (0.0-1.0 of the primary monitor) unless
``coord_type`` is ``"absolute"`` (whole pixels on the virtual desktop).

Supported actions
-----------------
    mouse.click     x, y, [coord_type, button, clicks, move_duration_ms]
    mouse.move      x, y, [coord_type, duration_ms]
    mouse.drag      from_x, from_y, to_x, to_y, [coord_type, duration_ms]
    mouse.scroll    direction, [amount, x, y, coord_type]
    keyboard.type   text, [interval_ms, mode = "paste" | "keystroke"]
    keyboard.hotkey keys
    window.focus / window.minimize / window.maximize / window.close
                    title_contains
    file.copy       source, destination, [overwrite]   (overwrite needs confirm)
    file.move       source, destination                (needs confirm)
    file.delete     path, [recursive]                  (needs confirm)

Security model (read this before deploying)
-------------------------------------------
* Action allowlist, strict schemas, bounded numeric ranges.
* Hotkey allowlist plus a blocklist of high-risk chords (Win+R, Alt+F4, ...).
* Forbidden screen zones are checked for every target AND every point on a
  planned movement path before the pointer moves.
* File operations are confined to ``file_roots`` (default: ``base_dir``), the
  audit log directory is protected, and delete/move require
  ``constraints.confirm == true``.
* Emergency failsafe: parking the pointer at (0, 0) aborts the action. During
  smooth movement the engine also aborts if the pointer is pulled away from the
  planned path (a human grabbed the mouse).
* Every event is written to ``logs/audit_ledger.log`` as hash-chained JSON.
  Typed text is never logged, only its length.

Known limitations
-----------------
* Timeouts are real, but a Python thread cannot be force-killed. The engine
  sets a cancel flag and injects an exception with
  ``PyThreadState_SetAsyncExc``; both take effect only when the worker is
  running Python code. A worker blocked inside a native call cannot be
  interrupted: the engine then reports the timeout, releases held inputs, and
  refuses further actions until that worker exits.
* Only TEXT clipboard contents survive ``keyboard.type`` in paste mode
  (pyperclip limitation). Images/files on the clipboard are not restored.
* The smooth cursor movement exists for visual clarity and reliable UI
  hover/drag handling. It is not designed to defeat bot detection, and it
  should not be used for that.
"""

_thread_ctx = threading.local()


def _raise_if_cancelled() -> None:
    """Cooperative cancellation point: raise if this worker's deadline passed."""
    event = getattr(_thread_ctx, "cancel_event", None)
    if event is not None and event.is_set():
        raise ActionAborted("Action cancelled after timeout.")


def _is_int(value: Any) -> bool:
    return isinstance(value, int) and not isinstance(value, bool)


def _is_num(value: Any) -> bool:
    return (
        isinstance(value, (int, float))
        and not isinstance(value, bool)
        and math.isfinite(value)
    )


def _is_within(path: Path, root: Path) -> bool:
    """True if ``path`` equals ``root`` or lies beneath it (both resolved)."""
    return path == root or root in path.parents


def _safe_id(value: Any) -> str:
    """Make an arbitrary action id safe for log records and file names."""
    cleaned = re.sub(r"[^A-Za-z0-9_.\-]", "_", str(value))[:64]
    return cleaned or "unknown"


def _extract_action_id(raw: Any) -> str:
    if isinstance(raw, dict) and isinstance(raw.get("action_id"), str):
        return _safe_id(raw["action_id"])
    return "unknown"

@dataclass(frozen=True)
class _Spec:
    """Declarative description of one validated field."""

    kind: str  # "bool" | "int" | "num" | "str" | "enum" | "keys"
    required: bool = False
    default: Any = None
    lo: Optional[float] = None
    hi: Optional[float] = None
    choices: tuple = ()
    min_len: int = 0
    max_len: int = 4096
    strip: bool = False


def _validate_value(name: str, value: Any, spec: _Spec) -> Any:
    """Validate one value against its spec and return the cleaned value."""
    kind = spec.kind
    if kind == "bool":
        if not isinstance(value, bool):
            raise ValueError(f"'{name}' must be a boolean.")
        return value
    if kind in ("int", "num"):
        ok = _is_int(value) if kind == "int" else _is_num(value)
        if not ok:
            raise ValueError(
                f"'{name}' must be {'an integer' if kind == 'int' else 'a finite number'}."
            )
        if spec.lo is not None and value < spec.lo:
            raise ValueError(f"'{name}' must be >= {spec.lo:g}.")
        if spec.hi is not None and value > spec.hi:
            raise ValueError(f"'{name}' must be <= {spec.hi:g}.")
        return value
    if kind == "str":
        if not isinstance(value, str):
            raise ValueError(f"'{name}' must be a string.")
        text = value.strip() if spec.strip else value
        if "\x00" in text:
            raise ValueError(f"'{name}' must not contain NUL characters.")
        if not spec.min_len <= len(text) <= spec.max_len:
            raise ValueError(
                f"'{name}' length must be between {spec.min_len} and {spec.max_len}."
            )
        return text
    if kind == "enum":
        if not isinstance(value, str) or value not in spec.choices:
            raise ValueError(f"'{name}' must be one of {list(spec.choices)}.")
        return value
    if kind == "keys":
        if not isinstance(value, list) or not all(isinstance(k, str) for k in value):
            raise ValueError(f"'{name}' must be a list of strings.")
        return list(value)
    raise ValueError(f"Unsupported schema kind '{kind}' for '{name}'.")


def _apply_schema(label: str, data: Any, schema: dict[str, _Spec]) -> dict[str, Any]:
    """Reject unknown keys, enforce required keys, fill defaults."""
    if not isinstance(data, dict):
        raise ValueError(f"'{label}' must be an object.")
    unknown = sorted(str(k) for k in set(data) - set(schema))
    if unknown:
        raise ValueError(f"Unknown {label} field(s): {unknown}.")
    cleaned: dict[str, Any] = {}
    for name, spec in schema.items():
        if name in data:
            cleaned[name] = _validate_value(name, data[name], spec)
        elif spec.required:
            raise ValueError(f"Missing required {label} field '{name}'.")
        else:
            cleaned[name] = spec.default
    return cleaned

@dataclass
class ActionContract:
    """Validated incoming action ("Aureon Action Contract")."""

    action_id: str
    type: str
    params: dict[str, Any] = field(default_factory=dict)
    constraints: dict[str, Any] = field(default_factory=dict)
    metadata: dict[str, Any] = field(default_factory=dict)

    @classmethod
    def from_dict(cls, data: Any) -> "ActionContract":
        """Strictly type-check ``data`` and build a contract.

        ``constraints`` is returned fully populated with defaults. ``params``
        is only checked to be an object here; the per-action schema is applied
        by the engine once the action type has passed the allowlist.

        Raises:
            ValueError: on any structural or type violation.
        """
        if not isinstance(data, dict):
            raise ValueError("Action must be a JSON object.")
        unknown = sorted(str(k) for k in set(data) - _CONTRACT_FIELDS)
        if unknown:
            raise ValueError(f"Unknown top-level field(s): {unknown}.")

        action_id = data.get("action_id")
        if not isinstance(action_id, str) or not _ACTION_ID_RE.fullmatch(action_id):
            raise ValueError(
                "'action_id' must be 1-64 chars of letters, digits, '_', '-' or '.'."
            )
        action_type = data.get("type")
        if not isinstance(action_type, str) or not _ACTION_TYPE_RE.fullmatch(action_type):
            raise ValueError("'type' must look like 'category.name' (e.g. 'mouse.click').")
        for key in ("params", "constraints", "metadata"):
            if key in data and not isinstance(data[key], dict):
                raise ValueError(f"'{key}' must be an object.")

        constraints = _apply_schema("constraints", data.get("constraints", {}), CONSTRAINT_SCHEMA)
        if constraints["requires_focus"] and not constraints["target_window"]:
            raise ValueError("constraints.requires_focus needs constraints.target_window.")

        return cls(
            action_id=action_id,
            type=action_type,
            params=dict(data.get("params", {})),
            constraints=constraints,
            metadata=dict(data.get("metadata", {})),
        )

_user32_handle: Any = None


def _user32() -> Any:
    """Private user32 handle (avoids clashing with other libraries' argtypes)."""
    global _user32_handle
    if _user32_handle is None:
        _user32_handle = ctypes.WinDLL("user32")  # type: ignore[attr-defined]
    return _user32_handle

class CoordinateEngine:
    """Coordinate conversion, DPI handling and multi-monitor lookup."""

    _dpi_aware: bool = False

    @staticmethod
    def relative_to_absolute(
        x: float, y: float, screen_width: int, screen_height: int
    ) -> tuple[int, int]:
        """Convert normalised (0.0-1.0) coordinates to pixel coordinates."""
        if not (_is_num(x) and _is_num(y) and 0.0 <= x <= 1.0 and 0.0 <= y <= 1.0):
            raise ValueError("Relative coordinates must be numbers between 0.0 and 1.0.")
        if screen_width <= 0 or screen_height <= 0:
            raise ValueError("Screen dimensions must be positive.")
        return (
            min(screen_width - 1, round(x * (screen_width - 1))),
            min(screen_height - 1, round(y * (screen_height - 1))),
        )

    @classmethod
    def enable_dpi_awareness(cls) -> bool:
        """Make this process DPI aware so input coordinates are physical pixels.

        Idempotent. Returns True when the process is DPI aware afterwards.
        pyautogui usually sets system awareness on import; this upgrades it to
        per-monitor awareness when the host has not already fixed a mode.
        """
        if not IS_WINDOWS:
            return False
        if cls._dpi_aware:
            return True
        try:
            shcore = ctypes.windll.shcore  # type: ignore[attr-defined]
            awareness = ctypes.c_int(-1)
            if shcore.GetProcessDpiAwareness(0, ctypes.byref(awareness)) == 0 and awareness.value != 0:
                cls._dpi_aware = True
                return True
            if shcore.SetProcessDpiAwareness(2) == 0:  # PROCESS_PER_MONITOR_DPI_AWARE
                cls._dpi_aware = True
                return True
        except Exception:
            _LOG.debug("shcore DPI awareness API unavailable.", exc_info=True)
        try:
            cls._dpi_aware = bool(_user32().SetProcessDPIAware())
        except Exception:
            cls._dpi_aware = False
        return cls._dpi_aware

    @staticmethod
    def get_dpi_scale() -> float:
        """Return the system DPI scale factor (1.0 == 96 DPI)."""
        if not IS_WINDOWS:
            return 1.0
        try:
            user32 = _user32()
            dpi = int(user32.GetDpiForSystem()) if hasattr(user32, "GetDpiForSystem") else 96
            return max(1.0, dpi / 96.0) if dpi else 1.0

                test_file = os.path.join(base_dir, "test_source.txt")
                with open(test_file, "w") as f:
                    f.write("This is a test file for Aureon.")
                
                action["type"] = "file.copy"
                action["params"] = {
                    "source": "test_source.txt",
                    "destination": "test_destination.txt",
                    "overwrite": True
                }
                action["constraints"]["confirm"] = True # نیاز به تایید
                print(f"📁  کپی فایل از test_source.txt به test_destination.txt...")

            elif choice == "10":
                action["type"] = "mouse.click"
                # مختصات مطلق در گوشه بالا-چپ (منطقه ممنوعه)
                action["params"] = {"x": 10, "y": 10, "coord_type": "absolute", "button": "left"}
                print("🛡️  تلاش برای کلیک در منطقه ممنوعه... (انتظار می‌رود خطا بدهد)")

            elif choice == "11":
                action["type"] = "file.delete"
                action["params"] = {"path": "test_source.txt"}
                # confirm را False می‌گذاریم تا خطا بدهد
                action["constraints"]["confirm"] = False
                print("🛡️  تلاش برای حذف بدون تایید... (انتظار می‌رود خطا بدهد)")

            else:
                print("❌ گزینه نامعتبر است.")
                continue

            # اجرای دستور
            print("\n⏳ در حال اجرا...")
            result = engine.execute(action)
            
            # نمایش خروجی
            print("\n📄 نتیجه اجرا:")
            print(json.dumps(result, indent=2, ensure_ascii=False))

            else:
                print(f"❌ عملیات شکست خورد. دلیل: {result.get('errors')}")
